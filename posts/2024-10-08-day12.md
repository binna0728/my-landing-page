```python
# ── Matplotlib 한글 폰트 설정: Windows 기준 ─────────────────────────
import matplotlib as mpl
import matplotlib.pyplot as plt
import matplotlib.font_manager as fm

# Windows에서 사용 가능한 한글 폰트 자동 찾기
def get_korean_font():
    """Windows에서 사용 가능한 한글 폰트를 찾아 반환"""
    # 일반적으로 Windows에 설치된 한글 폰트들
    korean_fonts = ['Malgun Gothic', 'NanumGothic', 'NanumBarunGothic', 'Gulim', 'Dotum', 'Batang']
    
    available_fonts = [f.name for f in fm.fontManager.ttflist]
    
    for font in korean_fonts:
        if font in available_fonts:
            return font
    
    # 한글 폰트를 찾지 못한 경우 기본 폰트 반환
    return 'DejaVu Sans'

# 한글 폰트 설정
korean_font = get_korean_font()
print(f"사용할 폰트: {korean_font}")

# Matplotlib 전역 설정
plt.rcParams.update({
    "font.size": 10,
    "font.family": korean_font
})
mpl.rcParams["axes.unicode_minus"] = False  # 음수 기호 깨짐 방지

print("✓ 한글 폰트 설정 완료!")
```
