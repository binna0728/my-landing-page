## 문제 4) enumerate — 1부터 번호 붙이기

리스트 컴프리헨션

[(i, v) for i, v in enumerate(items, start=1)]

는 사실 이 코드랑 똑같아 👇

indexed = []
for i, v in enumerate(items, start=1):
    indexed.append((i, v))   # 👈 리스트에 추가하는 함수: append()

실행 결과:
```python
items = [
    "a",
    "b",
    "c",
]

# TODO: enumerate(iterable, start=1) : 튜플 (번호, 값) 을 하나씩 만들어서 내보내는 함수
indexed = [
    (i, v) 
    for i, v in enumerate(items, start=1) # for문의 언패킹(unpacking) 파이썬이 알아서 (1, "a")를 풀어서 나눔
]

assert indexed == [(1,'a'), (2,'b'), (3,'c')]
print("✅ 문제 4 통과")
```

## 05. 함수 흐름제어2_실습과제_참고코드

## 실습: 환자 정보 관리 콘솔

1. 환자 정보 입력 → dict 만들어 리스트에 추가
2. 환자 목록 보기 → for와 enumerate(start=1)로 순번과 함께 출력
3. 이름으로 검색 (부분 일치) → for + 조건
4. 환자 정보 수정 (메모만 간단히) → 선택한 환자의 memo 업데이트
5. 환자 삭제 → 선택한 번호의 환자 제거
6. 간단 통계
  - 총 환자 수 (len)
  - 평균 나이 (sum / len, 환자 없으면 0)
  - 성별 분포 (count)
7. 데이터 초기화
8. 종료 → break

먼저 필요한 라이브러리를 불러옵니다:
```python
# 환자 정보를 받을 리스트 선언
patients = []

while True:
    print("------------------------------")
    cmd = int(input('메뉴 번호 선택(1~8): ').strip())

    # 1. 환자 정보 입력
    if cmd == 1:
        print('\n [환자 정보를 입력하세요.]')
        name = input("이름: ").strip()
        age = int(input("나이: ").strip())
        gender = input("성별(남성/여성): ").strip()

        if not all([name, age, gender]):
            print("\n입력이 올바르지 않습니다. 다시 시도하세요.")
            continue

        patients.append({'name': name, 'age': age, 'gender': gender})
        print('\n환자 정보가 입력되었습니다.')

    # 2. 환자 목록 보기
    elif cmd == 2:
        if not patients:
            print("\n환자 정보가 없습니다.")
        else:
            print("\n[환자 목록]")
            for idx, p in enumerate(patients, start=1):
                print(f"{idx}. 이름: {p.get('name')}, 나이: {p.get('age')}, 성별: {p.get('gender')}")

    # 3. 이름으로 검색
    elif cmd == 3:
        search_name = input("검색할 이름 입력: ").strip()
        matches = [f"{idx}. 이름: {p.get('name')}, 나이: {p.get('age')}, 성별: {p.get('gender')}"
                   for idx, p in enumerate(patients, start=1)
                   if search_name in p.get('name')]
        if not matches:
            print("\n검색 결과가 없습니다.")
        else:
            print("\n[검색 결과]")
            for m in matches:
                print(m)

    # 4. 환자 정보 수정
    elif cmd == 4:
        if not patients:
            print("\n환자 정보가 없습니다.")
        else:
            print("\n[수정 대상 환자 목록]")
            for idx, p in enumerate(patients, start=1):
                print(f"{idx}. 이름: {p.get('name')}, 나이: {p.get('age')}, 성별: {p.get('gender')}")

            num = int(input("수정할 환자의 번호 입력: "))
            if 1 <= num <= len(patients):
                new_name = input("새 이름: ").strip()
                new_age = int(input("새 나이: ").strip())
                new_gender = input("새 성별(남성/여성): ").strip()
                patients[num - 1]['name'] = new_name
                patients[num - 1]['age'] = new_age
                patients[num - 1]['gender'] = new_gender
                print("\n환자 정보가 수정되었습니다.")
            else:
                print("올바른 번호를 입력하세요.")

    # 5. 환자 삭제
    elif cmd == 5:
        if not patients:
            print("\n환자 정보가 없습니다.")
        else:
            print("\n[삭제 대상 환자 목록]")
            for idx, p in enumerate(patients, start=1):
                print(f"{idx}. 이름: {p.get('name')}, 나이: {p.get('age')}, 성별: {p.get('gender')}")

            num = int(input("삭제할 환자의 번호 입력: "))
            if 1 <= num <= len(patients):
                removed = patients.pop(num-1)
                print(f"\n{removed['name']} 환자 정보가 삭제되었습니다.")
            else:
                print("올바른 번호를 입력하세요.")

    # 6. 간단 통계
    elif cmd == 6:
        total = len(patients)
        if total == 0:
            avg_age = 0
            male = female = 0
        else:
            ages = [p['age'] for p in patients]
            avg_age = sum(ages) / total
            male = sum(1 for p in patients if p['gender'] == '남성')
            female = sum(1 for p in patients if p['gender'] == '여성')

        print("\n[통계]")
        print(f"총 환자 수: {total}")
        print(f"평균 나이: {avg_age:.1f}")
        print(f"성별 분포: 남 {male}명 / 여 {female}명")

    # 7. 데이터 초기화
    elif cmd == 7:
        confirm = input("\n모든 데이터를 초기화할까요? (Y/N): ").strip().upper()
        if confirm == 'Y':
            patients.clear()
            print("\n데이터가 초기화되었습니다.")
        else:
            print("취소되었습니다.")

    # 8. 종료
    elif cmd == 8:
        print("\n프로그램을 종료합니다.")
        break

    else:
        print("올바른 메뉴 번호를 입력하세요.")
```
