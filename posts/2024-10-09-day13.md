### draw_io_er_diagram.py

```python
"""
draw.io ER Diagram XML 파일 생성 스크립트
학생(Student), 수업(Class), 교수(Professor) 엔터티와
n:m, n:1 관계를 자동으로 생성합니다.
"""

import xml.etree.ElementTree as ET
import os
from pathlib import Path


def create_drawio_xml():
    """draw.io 호환 XML 파일 생성"""

    # draw.io 기본 구조
    mxfile = ET.Element('mxfile', {
        'host': 'app.diagrams.net',
        'modified': '2025-10-27T02:00:00.000Z',
        'agent': 'Mozilla/5.0',
        'etag': 'generated',
        'version': '1.0',
        'type': 'device'
    })

    diagram = ET.SubElement(mxfile, 'diagram', {'id': 'diagram1', 'name': 'ER Diagram'})
    mxGraphModel = ET.SubElement(diagram, 'mxGraphModel', {
        'dx': '1200',
        'dy': '800',
        'grid': '1',
        'gridSize': '10',
        'guides': '1',
        'tooltips': '1',
        'connect': '1',
        'arrows': '1',
        'fold': '1',
        'page': '1',
        'pageScale': '1',
        'pageWidth': '1200',
        'pageHeight': '800',
        'math': '0',
        'shadow': '0'
    })

    root = ET.SubElement(mxGraphModel, 'root')

    # Cell 0: 기본 셀
    ET.SubElement(root, 'mxCell', {'id': '0'})

    # Cell 1: 캔버스 배경
    ET.SubElement(root, 'mxCell', {
        'id': '1',
        'parent': '0',
        'value': ''
    })

    cell_id = 2

    # 엔터티 데이터: (이름, x, y, width, height)
    entities = [
        ('학생', 100, 80, 100, 60),
        ('수업', 450, 80, 100, 60),
        ('교수', 800, 80, 100, 60),
        ('수강', 275, 250, 100, 60),
    ]

    entity_cells = {}

    # 엔터티 셀 생성
    print("[*] 엔터티 생성 중...")
    for name, x, y, w, h in entities:
        cell_id_str = str(cell_id)
        entity_cells[name] = cell_id_str

        mxGeometry = ET.SubElement(
            ET.SubElement(root, 'mxCell', {
                'id': cell_id_str,
                'value': name,
                'style': 'shape=rectangle;strokeWidth=2;fillColor=#ffffff;rounded=1;fontSize=12;fontStyle=1;align=center;verticalAlign=middle;',
                'vertex': '1',
                'parent': '1'
            }),
            'mxGeometry',
            {'x': str(x), 'y': str(y), 'width': str(w), 'height': str(h), 'as': 'geometry'}
        )

        print(f"   [+] {name} 추가 (위치: {x}, {y})")
        cell_id += 1

    # 관계선 데이터: (from, to, label)
    relationships = [
        ('학생', '수강', '1:N'),
        ('수강', '수업', '1:N'),
        ('교수', '수업', 'N:1'),
    ]

    # 관계선 생성
    print("\n[*] 관계선 생성 중...")
    for from_entity, to_entity, label in relationships:
        source_id = entity_cells.get(from_entity)
        target_id = entity_cells.get(to_entity)

        if source_id and target_id:
            cell_id_str = str(cell_id)

            mxGeometry = ET.SubElement(
                ET.SubElement(root, 'mxCell', {
                    'id': cell_id_str,
                    'value': label,
                    'style': 'edgeStyle=orthogonalEdgeStyle;rounded=1;fontSize=10;',
                    'edge': '1',
                    'parent': '1',
                    'source': source_id,
                    'target': target_id
                }),
                'mxGeometry',
                {'relative': '1', 'as': 'geometry'}
            )

            print(f"   [+] {from_entity} -> {to_entity} [{label}]")
            cell_id += 1

    # XML을 파일로 저장
    output_path = Path("D:\\00.오즈코딩 실습\\13일차\\er_diagram.drawio")
    output_path.parent.mkdir(parents=True, exist_ok=True)

    tree = ET.ElementTree(mxfile)
    ET.indent(tree, space="  ")
    tree.write(output_path, encoding='utf-8', xml_declaration=True)

    return str(output_path)


def open_in_drawio(file_path):
    """생성된 파일을 draw.io에서 열기"""
    import webbrowser
    from urllib.parse import quote

    # draw.io 온라인 에디터에서 파일 열기
    file_url = Path(file_path).as_uri()
    drawio_url = f"https://app.diagrams.net/?import={quote(file_url)}"

    print(f"\n[*] draw.io 온라인 에디터에서 열기...")
    print(f"[*] URL: {drawio_url}")

    webbrowser.open(drawio_url)


def main():
    print("""
draw.io ER Diagram XML 파일 생성
====================================================
    """)

    print("\n" + "="*70)
    print("  draw.io ER Diagram 생성")
    print("="*70)

    print("\n[1단계] XML 파일 생성")
    print("-"*70)

    try:
        file_path = create_drawio_xml()
        print(f"\n[+] 파일 생성 완료: {file_path}")

        print("\n[2단계] draw.io에서 열기")
        print("-"*70)
        open_in_drawio(file_path)

        print("\n[*] 팁:")
        print("   - 브라우저에서 다이어그램을 볼 수 있습니다")
        print("   - Ctrl+S: 다이어그램 저장")
        print("   - 더블클릭: 텍스트 편집")
        print("   - 선을 더블클릭: 라벨 수정")

        print("\n" + "="*70)
        print("[+] ER Diagram 생성 완료!")
        print("="*70)

    except Exception as e:
        print(f"\n[-] 오류 발생!")
        print(f"[-] {e}")
        import traceback
        traceback.print_exc()


if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print("\n[*] 프로그램 중단됨")
```
