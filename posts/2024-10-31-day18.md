for col in na_cols:
    if df[col].dtype == 'object':  # 범주형
        mode_vals = df[col].mode(dropna=True)
        if not mode_vals.empty:
            df[col] = df[col].fillna(mode_vals.iloc[0])
    else:  # 숫자형
        median_val = df[col].median(skipna=True)
        df[col] = df[col].fillna(median_val)
