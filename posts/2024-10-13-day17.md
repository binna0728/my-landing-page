### convert_encoding.py

```python
"""
CSV 인코딩 변환 도구
- 대상 파일을 CP949(또는 지정 인코딩)에서 UTF-8로 변환
- 원본은 .bak로 백업
"""
from pathlib import Path

TARGETS = [
    "drugs_v2.csv",
    "treatments_v2.csv",
]


def convert_file(path: Path, src_encoding: str = "cp949", dst_encoding: str = "utf-8") -> bool:
    backup = path.with_suffix(path.suffix + ".bak")
    data = path.read_bytes()
    try:
        text = data.decode(src_encoding)
    except Exception as e:
        print(f"[SKIP] {path.name}: {src_encoding} 디코딩 실패 - {e}")
        return False

    # 백업 저장 후 UTF-8로 덮어쓰기
    if not backup.exists():
        backup.write_bytes(data)
    path.write_text(text, encoding=dst_encoding, newline="\n")
    print(f"[OK] {path.name}: {src_encoding} -> {dst_encoding} 변환, 백업={backup.name}")
    return True


def main():
    base = Path(__file__).resolve().parent
    for name in TARGETS:
        p = base / name
        if not p.exists():
            print(f"[MISS] {name}: 파일 없음")
            continue
        convert_file(p, src_encoding="cp949", dst_encoding="utf-8")


if __name__ == "__main__":
    main()
```


### models.py

```python
"""
Tortoise ORM 모델 정의
- CLAUDE.md의 3NF 스키마를 기준으로 작성
"""
from tortoise import fields
from tortoise.models import Model


class Patient(Model):
    """환자 정보(patients)"""

    id = fields.IntField(pk=True)
    name = fields.CharField(max_length=100)
    gender = fields.CharField(max_length=1)
    birth_date = fields.DateField()

    health_metrics: fields.ReverseRelation["HealthMetric"]
    treatments: fields.ReverseRelation["Treatment"]

    class Meta:
        table = "patients"


class Drug(Model):
    """의약품 정보(drugs)"""

    id = fields.IntField(pk=True)
    drug_code = fields.CharField(max_length=50)
    drug_name = fields.CharField(max_length=100)
    drug_company = fields.CharField(max_length=100)
    company_phone = fields.CharField(max_length=20)

    treatments: fields.ReverseRelation["Treatment"]

    class Meta:
        table = "drugs"


class HealthMetric(Model):
    """건강 지표(health_metrics)"""

    id = fields.IntField(pk=True)
    patient: fields.ForeignKeyRelation[Patient] = fields.ForeignKeyField(
        "models.Patient", related_name="health_metrics", on_delete=fields.CASCADE
    )
    date = fields.DateField()
    weight = fields.IntField()
    height = fields.IntField()
    systolic = fields.IntField()
    diastolic = fields.IntField()
    glucose = fields.IntField()
    heart_rate = fields.IntField()

    class Meta:
        table = "health_metrics"


class Treatment(Model):
    """처방 정보(treatments)"""

    id = fields.IntField(pk=True)
    patient: fields.ForeignKeyRelation[Patient] = fields.ForeignKeyField(
        "models.Patient", related_name="treatments", on_delete=fields.CASCADE
    )
    date = fields.DateField()
    drug: fields.ForeignKeyRelation[Drug] = fields.ForeignKeyField(
        "models.Drug", related_name="treatments", on_delete=fields.CASCADE
    )
    dose = fields.CharField(max_length=50)
    prescription = fields.CharField(max_length=100)

    class Meta:
        table = "treatments"
```
