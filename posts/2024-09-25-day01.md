### 연습 2: 건강 점수 계산

`calculate_health_score` 함수를 정의했습니다:
```python
def calculate_health_score(temp, bp):
    """
    체온과 혈압을 기반으로 건강 점수 계산
    - 체온 정상(36.0~37.5): 50점
    - 혈압 정상(<130): 50점
    """
    score = 0
    
    # 체온 점수
    if 36.0 <= temp <= 37.5:
        score += 50
    
    # 혈압 점수
    if bp < 130:
        score += 50
    
    return score

# 주간 건강 점수 계산
total_score = 0
for day, data in health_data.items():
    daily_score = calculate_health_score(data["체온"], data["혈압"])
    total_score += daily_score
    print(f"{day}: {daily_score}점")

avg_score = total_score / len(health_data)
print(f"\n주간 평균 건강 점수: {avg_score:.1f}점")

if avg_score >= 80:
    print("건강 상태: 매우 좋음 😊")
elif avg_score >= 60:
    print("건강 상태: 양호 🙂")
else:
    print("건강 상태: 주의 필요 ⚠️")
```

## 📌 과제 난이도 하 - 체온 판별 함수 정리
사용자가 입력한 체온을 네 단계(낮음/정상/미열/고열)로 분류하고 안내 메시지를 제공합니다.

`classify_temperature` 함수를 정의했습니다:
```python
from typing import Iterable, List, Tuple

TEMP_MESSAGES = {
    "고열": "🔥 고열입니다.",
    "미열": "🤒 미열입니다.",
    "정상": "✅ 정상입니다.",
    "낮음": "⚠️ 낮은 체온입니다.",
}

def classify_temperature(temperature: float) -> Tuple[str, str]:
    """체온에 따라 (상태, 메시지) 튜플을 반환합니다."""
    if temperature >= 38.0:
        status = "고열"
    elif temperature >= 37.5:
        status = "미열"
    elif temperature >= 36.5:
        status = "정상"
    else:
        status = "낮음"
    return status, TEMP_MESSAGES[status]
```

실행 결과:
```python
example_temperatures = [36.8, 37.8, 38.2, 32.1]

for value in example_temperatures:
    _, message = classify_temperature(value)
    print(f"체온을 입력하세요: {value:.1f} -> {message}")
```

### 더 성장하기: map()으로 여러 체온을 한 번에 판별

`bulk_classify_temperatures` 함수를 정의했습니다:
```python
def bulk_classify_temperatures(temperatures: Iterable[float]) -> List[Tuple[float, str, str]]:
    """map()을 사용해 체온과 판별 결과를 묶어 반환합니다."""
    return list(map(lambda value: (value, *classify_temperature(value)), temperatures))

for temp, status, message in bulk_classify_temperatures(example_temperatures):
    print(f"{temp:.1f}°C -> {status} / {message}")
```

## 💻 과제 난이도 상 - 바이탈 사인 판별 프로그램
날짜, 혈압, 혈당 입력을 받아 정상/주의/위험 여부를 안내합니다.

`classify_blood_pressure` 함수를 정의했습니다:
```python
from datetime import datetime

def classify_blood_pressure(value: int) -> str:
    """혈압을 정상/주의/위험으로 구분합니다."""
    if value >= 140:
        return "위험"
    if value >= 120:
        return "주의"
    return "정상"

def classify_blood_sugar(value: int) -> str:
    """혈당을 정상/주의/위험으로 구분합니다."""
    if value >= 126:
        return "위험"
    if value >= 100:
        return "주의"
    return "정상"

def _validate_measurement(name: str, value: int) -> int:
    if not 0 <= value < 300:
        raise ValueError(f"{name}은 0 이상 300 미만으로 입력하세요: {value}")
    return value

def evaluate_vital_signs(date: str, blood_pressure: int, blood_sugar: int) -> str:
    """날짜 문자열과 혈압·혈당을 받아 판별 메시지를 생성합니다."""
    datetime.strptime(date, "%Y-%m-%d")
    bp = _validate_measurement("혈압", blood_pressure)
    sugar = _validate_measurement("혈당", blood_sugar)
    bp_status = classify_blood_pressure(bp)
    sugar_status = classify_blood_sugar(sugar)
    return f"[{date}] 혈압: {bp} ({bp_status}), 혈당: {sugar} ({sugar_status})"
```

실행 결과:
```python
print(evaluate_vital_signs("2025-09-09", 135, 90))

```
