아래 `log_decorator`와 `greet`가 있습니다.  
`@log_decorator`를 **사용하지 말고**, 동일한 효과가 나도록 코드를 작성하세요.

데코레이터는 **'함수를 받아서 새로운 함수를 반환하는 함수'**다. 


@log_decorator 라고 쓰는 것은 사실 greet = log_decorator(greet) 와 같다.

`log_decorator` 함수를 정의했습니다:
```python
def log_decorator(func):
    def wrapper():
        print("[LOG] 함수 시작")
        func()
        print("[LOG] 함수 끝")
    return wrapper

def greet():
    print("안녕하세요!")

# TODO: 데코레이터 문법(@...) 대신 함수 대입으로 동일 효과 만들기

# 테스트 (시각 확인)
```


## 3) 캡슐화 + `@property`

요구사항
- MapleCharacter 클래스 만들기
- __hp는 외부에서 직접 접근 불가
- get_hp(), set_hp() 메서드로 HP를 안전하게 관리
- @property / @hp.setter 로 같은 기능을 속성 접근 형태로도 구현

`__init__` 함수를 정의했습니다:
```python
class MapleCharacter:
    def __init__(self, name):
        self.name = name
        self.__hp = 100   # private 변수

    # ✅ 1️⃣ getter
    def get_hp(self):
        return self.__hp

    # ✅ 2️⃣ setter
    def set_hp(self, value):
        if value < 0:
            self.__hp = 0
        elif value > 1000:
            self.__hp = 1000
        else:
            self.__hp = value

    # ✅ 3️⃣ property getter
    @property
    def hp(self):
        return self.__hp

    # ✅ 4️⃣ property setter
    @hp.setter  
    def hp(self, value):
        if value < 0:
            self.__hp = 0
        elif value > 1000:
            self.__hp = 1000
        else:
            self.__hp = value


# 🧪 테스트 (assert 자동 채점)
m = MapleCharacter("루카")

# ① getter/setter 테스트
m.set_hp(1200)
assert m.get_hp() == 1000, "❌ set_hp(1200) → HP는 1000이어야 합니다"

m.set_hp(-10)
assert m.get_hp() == 0, "❌ set_hp(-10) → HP는 0이어야 합니다"

m.set_hp(500)
assert m.get_hp() == 500, "❌ set_hp(500) → HP는 500이어야 합니다"

# ② property 테스트
m.hp = 2000
assert m.hp == 1000, "❌ m.hp=2000 → HP는 1000이어야 합니다"

m.hp = -50
assert m.hp == 0, "❌ m.hp=-50 → HP는 0이어야 합니다"

m.hp = 750
assert m.hp == 750, "❌ m.hp=750 → HP는 750이어야 합니다"

print("✅ 3) 통과")
```

**출력:**
```
✅ 3) 통과
```

```python
def time_decorator(func):
    def wrapper():
        start = time.time()
        func()
        print('함수 실행시간(초):', time.time()-start)
    return wrapper
```

```python
    server = '크로아'

        self.name = name
        self.STR = STR
        self.LUK = LUK

    @property

    @hp.setter

        if(amount<0):
            self.__hp = 0
        else:
            self.__hp = amount
            

    @property
    def level(self):
        print(self.__level)

    def get_exp(self):

        if(self.__exp >= 100):
```

### Character.py

```python
import random

characters = []

class Character:
    total = 0
    def __init__(self, name, str_, dex, int_, luk):
        self.name = name
        self.str = str_
        self.dex = dex
        self.int = int_
        self.luk = luk
        self.level = 1
        self.hp = 100
        self.mp = 100
        self.exp = 0
        self.money = 0
        Character.total += 1

print(">>> RUNNING:", __file__)

while True:
    print("\n1. 캐릭터 생성")
    print("2. 캐릭터 목록 보기")
    print("3. 종료")
    menu = input("메뉴를 선택하세요: ").strip()

    if menu == '1':
        name = input("생성할 캐릭터의 이름을 입력하세요: ").strip()
        new_str = random.randint(4, 8)
        new_dex = random.randint(4, 8)
        new_int = random.randint(4, 8)
        new_luk = random.randint(4, 8)

        print("\n능력치가 결정되었습니다!")
        print(f"이름: {name}")
        print(f"STR: {new_str}")
        print(f"DEX: {new_dex}")
        print(f"INT: {new_int}")
        print(f"LUK: {new_luk}")

        new_char = Character(name, new_str, new_dex, new_int, new_luk) 
        characters.append(new_char)                                     
        print(f"\n{name}님, 캐릭터 생성이 완료되었습니다! (총 {Character.total}명)")

    elif menu == '2':
        print("\n--- [생성된 캐릭터 목록] ---")
        if not characters:
            print("아직 생성된 캐릭터가 없습니다.")
        else:
            for i, c in enumerate(characters, 1):
                print(f"{i}. 이름: {c.name}, STR: {c.str}, DEX: {c.dex}, INT: {c.int}, LUK: {c.luk}")
        print("--------------------------")

    elif menu == '3':
        print("프로그램을 종료합니다.")
        break
    else:
        print("잘못된 입력입니다.")
```
