## 실습 코드


### app.py

```python
import asyncio
from datetime import datetime, timedelta

from tortoise import Tortoise
from models import Patient, Doctor, Appointment
from config import TORTOISE_ORM


async def main():
    # 1. Tortoise ORM 초기화 및 스키마 생성
    await Tortoise.init(config=TORTOISE_ORM)
    await Tortoise.generate_schemas()

    print("=" * 50)
    print("CREATE - 데이터 생성")
    print("=" * 50)

    # 2. CREATE - 의사(Doctor) 생성
    dr_lee = await Doctor.create(name="이병헌", specialty="내과")
    dr_park = await Doctor.create(name="박명수", specialty="소아과")
    print(f"의사 생성: {dr_lee}, {dr_park}")

    # 3. CREATE - 환자(Patient) 생성
    p_lee = await Patient.create(name="이동원", birth_date="1993-01-01")
    p_kim = await Patient.create(name="김민찬", birth_date="2000-05-10")
    print(f"환자 생성: {p_lee}, {p_kim}")

    # 4. CREATE - 예약(Appointment) 생성
    appt1 = await Appointment.create(
        patient=p_lee, doctor=dr_lee,
        reserved_at=datetime.now() + timedelta(days=1),
        note="초진 예약"
    )
    appt2 = await Appointment.create(
        patient=p_lee, doctor=dr_park,
        reserved_at=datetime.now() + timedelta(days=2),
        note="소아과 상담"
    )
    print(f"예약 생성: {appt1}, {appt2}")

    print("\n" + "=" * 50)
    print("READ - 데이터 조회")
    print("=" * 50)

    # 5. READ - 단건 조회
    one_doctor = await Doctor.get(id=dr_lee.id)
    print(f"단건 의사 조회: {one_doctor}")

    # 6. READ - 조건 조회 (filter)
    recent_patients = await Patient.filter(birth_date__gte="1990-01-01").order_by("name")
    print(f"1990년 이후 출생 환자: {[str(p) for p in recent_patients]}")

    # 7. READ - 관계 조회 (select_related)
    rows = await Appointment.all().select_related("patient", "doctor")
    print("\n예약 목록 (관계 포함):")
    for a in rows:
        print(f"  [예약 {a.id}] {a.patient.name} → {a.doctor.name} ({a.note})")

    # 8. READ - 역방향 관계 조회 (prefetch_related)
    lee_with_appts = await Patient.get(id=p_lee.id).prefetch_related("appointments")
    print(f"\n{lee_with_appts.name}의 예약 수: {len(lee_with_appts.appointments)}")

    print("\n" + "=" * 50)
    print("UPDATE - 데이터 수정")
    print("=" * 50)

    # 9. UPDATE - 의사 정보 수정
    dr_lee.specialty = "가정의학과"
    await dr_lee.save()
    print(f"의사 정보 수정: {dr_lee}")

    # 10. UPDATE - 예약 정보 수정
    appt1.note = "증상 업데이트: 기침/두통/몸살"
    appt1.reserved_at = appt1.reserved_at + timedelta(days=1)
    await appt1.save()
    print(f"예약 정보 수정: {appt1} - {appt1.note}")

    print("\n" + "=" * 50)
    print("DELETE - 데이터 삭제")
    print("=" * 50)

    # 11. DELETE - 예약 삭제
    await appt2.delete()
    print(f"예약 삭제: appt2 (id={appt2.id})")

    # 12. DELETE - 환자 삭제 (CASCADE로 연관 예약도 삭제됨)
    await p_lee.delete()
    print(f"환자 삭제: {p_lee.name} (CASCADE로 연관 예약도 삭제)")

    # 13. DELETE - 의사 삭제
    await dr_park.delete()
    print(f"의사 삭제: {dr_park.name}")

    print("\n" + "=" * 50)
    print("최종 데이터 확인")
    print("=" * 50)

    # 14. 최종 데이터 확인
    remaining_doctors = await Doctor.all()
    remaining_patients = await Patient.all()
    remaining_appts = await Appointment.all()

    print(f"남은 의사: {[str(d) for d in remaining_doctors]}")
    print(f"남은 환자: {[str(p) for p in remaining_patients]}")
    print(f"남은 예약: {[str(a) for a in remaining_appts]}")

    # 15. 연결 종료
    await Tortoise.close_connections()
    print("\n프로그램 종료")


if __name__ == "__main__":
    asyncio.run(main())
```


### config.py

```python
import os
from pathlib import Path
from dotenv import load_dotenv

if Path(".env").exists():
    load_dotenv(".env")

MYSQL_USER = os.getenv("MYSQL_USER", "root")
MYSQL_PASSWORD = os.getenv("MYSQL_PASSWORD", "password")
MYSQL_HOST = os.getenv("MYSQL_HOST", "127.0.0.1")
MYSQL_PORT = os.getenv("MYSQL_PORT", "3306")
MYSQL_DB = os.getenv("MYSQL_DB", "demo_db")

DB_URL = f"mysql://{MYSQL_USER}:{MYSQL_PASSWORD}@{MYSQL_HOST}:{MYSQL_PORT}/{MYSQL_DB}?charset=utf8mb4"

TORTOISE_ORM = {
    "connections": {
        "default": DB_URL
    },
    "apps": {
        "models": {
            "models": ["models", "aerich.models"],
            "default_connection": "default",
        },
    },
}
```
